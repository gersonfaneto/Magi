* Filesystems
/dev/nvme0n1: WDC PC SN730 256GB (single-disk btrfs)
#+begin_src sh
parted /dev/nvme0n1 -- mklabel gpt
parted /dev/nvme0n1 -- mkpart EFI fat32 1MiB 1GiB
parted /dev/nvme0n1 -- mkpart root 1GiB 100%
parted /dev/nvme0n1 -- set 1 boot on
mkfs.fat -F32 /dev/nvme0n1p1
mkfs.btrfs -L root /dev/nvme0n1p2
mount /dev/nvme0n1p2 /mnt
btrfs subvolume create /mnt/home
btrfs subvolume create /mnt/nix
#+end_src

Swap lives on its own partition, /dev/nvme0n1p3.
